# 2026-2027 Aidat Roster Importer — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Parse the aidat workbook's `2026 - 2027` sheet into a reviewable CSV and, after human review, upload the roster + forward fee schedule into local Supabase.

**Architecture:** Pure logic modules under `scripts/seed/lib/` (aidat cell parsing, program inference, phone-match scoring, import-plan building) are unit-tested with vitest and have zero IO. Two thin CLIs (`parse-aidat.ts`, `upload-aidat.ts`) wire the pure modules to the filesystem and Supabase, mirroring the existing `parse-excel.ts` / `upload.ts` seed pattern.

**Tech Stack:** TypeScript, `tsx` (CLI runner), `xlsx` (SheetJS), `@supabase/supabase-js` (service role), vitest.

## Global Constraints

- **Money:** DB stores **lira** (`numeric(12,2)`); compute in **kuruş** and convert at the boundary using `lib/money.ts` (`tlToKurus`, `kurusToTl`) and `lib/installments.ts` (`generateInstallments`). Never hand-roll installment division.
- **Programs enum:** `tam_gun | yarim_gun | uc_gun | play_grup` (from `lib/constants.ts` `PROGRAMS`).
- **Item types enum:** `aidat | kayit_ucreti | servis | yemek | diger`.
- **Season constant:** `2026-2027`.
- **First due date anchor:** `2026-09-05`; aidat/servis = 10 monthly installments (Eyl 2026→Haz 2027), all `status: 'bekliyor'`; kitap = 1 installment.
- **Phone format:** E.164 TR mobile via `lib/phone.ts` `normalizeTurkishPhone(raw): string | null`.
- **Privacy:** `scripts/seed/out/` is gitignored. No real names/phones in scripts, tests, or commits. Reports reference `source_row` only; never log phone numbers.
- **DB safety:** `upload-aidat.ts` is **dry-run by default**; a real run requires an explicit non-`--dry-run` invocation and targets local Supabase only.
- **Turkish comments** to match the existing seed scripts' style.

// scripts/seed/lib/aidat-parse.ts
/**
 * Aidat çalışma kitabının "2026 - 2027" sayfasındaki bir satırı, insan onayı için
 * CSV'ye yazılacak yapıya çevirir. Saf fonksiyonlar — dosya/DB erişimi YOK.
 *
 * KOLONLAR (2026 - 2027 sayfası, 0-index):
 *   0 ÖĞRENCİ ADI VE SOYADI | 1 ANNE & BABA ADI | 2 KİTAP - KIRTASİYE
 *   3 AİDAT TUTARI | 4 (boş) | 5 SERVİS TUTARI  (6..17 aylık defter — YOK SAYILIR)
 */
import { PROGRAMS, type Program } from "../../lib/constants";

export const SEASON = "2026-2027";

const ANNUAL_LUMP_THRESHOLD = 100000; // bundan büyük tek sayı yıllık kabul edilir
const TAM_MONTHLY_THRESHOLD = 25000; // aylık >= bu -> tam_gun (tek etiketli veri noktası: 25000=tam, 23000=yarım)

export type ParsedAidat = {
  monthlyTl: number | null;
  program: Program | null;
  programNote: string;
  note: string;
};

/** Bir sayı metnini TL'ye çevirir ("45.000" / "45000" / 45000 -> 45000). Aksi halde null. */
export function parseTl(raw: unknown): number | null {
  if (typeof raw === "number") return Number.isFinite(raw) && raw > 0 ? raw : null;
  const s = String(raw ?? "").trim();
  if (!s) return null;
  // "23.000" (binlik nokta) ve "23,5" (ondalık virgül) karışık olabilir; sadece rakamları al.
  const digits = s.replace(/[^\d]/g, "");
  if (!digits) return null;
  const n = Number(digits);
  return Number.isFinite(n) && n > 0 ? n : null;
}

function inferProgram(monthlyTl: number | null, hasTamText: boolean): Program | null {
  if (hasTamText) return "tam_gun";
  if (monthlyTl === null) return null;
  return monthlyTl >= TAM_MONTHLY_THRESHOLD ? "tam_gun" : "yarim_gun";
}

export function parseAidatCell(rawAidat: unknown): ParsedAidat {
  // Sayısal hücre: büyükse yıllık (÷10), küçükse aylık.
  if (typeof rawAidat === "number" && rawAidat > 0) {
    const monthly = rawAidat >= ANNUAL_LUMP_THRESHOLD ? rawAidat / 10 : rawAidat;
    return { monthlyTl: monthly, program: inferProgram(monthly, false), programNote: "guess: aidat", note: "" };
  }
  const s = String(rawAidat ?? "").trim();
  if (!s) return { monthlyTl: null, program: null, programNote: "", note: "" };

  // "23.000 Y - 28.000 TAM" gibi metin: TAM figürünü aylık al, ham metni not'a koy.
  const hasTam = /TAM/i.test(s);
  const nums = [...s.matchAll(/\d[\d.]*/g)].map((m) => Number(m[0].replace(/\./g, ""))).filter((n) => n > 0);
  if (hasTam && nums.length > 0) {
    const monthly = Math.max(...nums); // TAM figürü genelde en büyüğü
    return { monthlyTl: monthly, program: "tam_gun", programNote: "guess: 'TAM' metni", note: s };
  }
  // ÜCRETSİZ / burslu / '?' vb. — tutar/program yok, ham metni sakla.
  return { monthlyTl: null, program: null, programNote: "", note: s };
}

export type RosterRow = {
  source_row: number;
  full_name: string;
  parent_name: string;
  program: Program | null;
  program_note: string;
  season: string;
  aidat_monthly: number | null;
  servis_monthly: number | null;
  kitap_amount: number | null;
  note: string;
};

export function buildRosterRow(excelRow: unknown[], sourceRow: number): RosterRow | null {
  const full_name = String(excelRow[0] ?? "").trim();
  if (!full_name) return null;
  const parent_name = String(excelRow[1] ?? "").trim();
  const parsed = parseAidatCell(excelRow[3]);
  const kitap_amount = parseTl(excelRow[2]);
  const servis_monthly = parseTl(excelRow[5]);

  const notes: string[] = [];
  if (parsed.note) notes.push(`aidat_ham: ${parsed.note}`);

  return {
    source_row: sourceRow,
    full_name,
    parent_name,
    program: parsed.program,
    program_note: parsed.programNote,
    season: SEASON,
    aidat_monthly: parsed.monthlyTl,
    servis_monthly,
    kitap_amount,
    note: notes.join(" | "),
  };
}

// PROGRAMS importunun kullanıldığından emin ol (tip güvenliği + geçerli enum kontrolü ileride).
void PROGRAMS;
```

// scripts/seed/lib/phone-recovery.ts
/**
 * Telefon kurtarma SKORLAMASI — saf fonksiyonlar. Havuzun (pool) dosyalardan
 * yüklenmesi CLI tarafında (parse-aidat.ts) yapılır; burada yalnız eşleştirme mantığı.
 *
 * Skor: soyadı geçiyor +3, her ebeveyn adı +2, çocuk adı +1. >=3 kabul.
 * Güven: >=6 high, 4-5 medium, 3 low. Sohbet bloğu: soyadı VE bir ebeveyn adı aynı
 * blokta geçmeli (aksi halde yanlış numara kapılır).
 */
export function normalizeName(s: string): string {
  return String(s ?? "")
    .replace(/İ/g, "I").replace(/ı/g, "I").replace(/i/g, "I")
    .toUpperCase()
    .replace(/[^A-ZÇĞÖŞÜ ]/g, " ")
    .replace(/\s+/g, " ")
    .trim();
}
export function nameTokens(s: string): string[] {
  return normalizeName(s).split(" ").filter((t) => t.length >= 3);
}

export type PoolEntry = { name: string; phone: string; source: string };
export type ChatBlock = { text: string; phones: string[] };
export type Query = { surname: string; first: string; parentTokens: string[] };
export type PhoneMatch = { phone: string; confidence: "high" | "medium" | "low"; source: string };

export function buildQuery(fullName: string, parents: string): Query {
  const t = nameTokens(fullName);
  const parentTokens = parents.split(/[&\-,]/).flatMap((p) => nameTokens(p));
  return { surname: t[t.length - 1] ?? "", first: t[0] ?? "", parentTokens };
}

function confidenceFor(score: number): PhoneMatch["confidence"] {
  if (score >= 6) return "high";
  if (score >= 4) return "medium";
  return "low";
}

export function scorePhone(query: Query, pool: PoolEntry[], chat: ChatBlock[]): PhoneMatch | null {
  const best: Record<string, { score: number; source: string }> = {};
  const consider = (phone: string, score: number, source: string) => {
    if (score < 3) return;
    if (!best[phone] || best[phone].score < score) best[phone] = { score, source };
  };

  for (const e of pool) {
    let sc = 0;
    if (query.surname && e.name.includes(query.surname)) sc += 3;
    for (const p of query.parentTokens) if (p && e.name.includes(p)) sc += 2;
    if (query.first && e.name.includes(query.first)) sc += 1;
    consider(e.phone, sc, e.source);
  }
  for (const b of chat) {
    if (query.surname && b.text.includes(query.surname) && query.parentTokens.some((p) => p && b.text.includes(p))) {
      for (const ph of b.phones) consider(ph, 4, "chat-block");
    }
  }

  const ranked = Object.entries(best).sort((a, b) => b[1].score - a[1].score);
  if (ranked.length === 0) return null;
  const [phone, info] = ranked[0];
  return { phone, confidence: confidenceFor(info.score), source: info.source };
}

// scripts/seed/parse-aidat.ts
/**
 * Aidat çalışma kitabının "2026 - 2027" sayfasını okuyup insan onayı için
 * out/students-2026-2027.csv üretir. DB'ye YAZMAZ (spec §9.3 akışı: parse→onay→upload).
 *
 * Kullanım:
 *   npx tsx scripts/seed/parse-aidat.ts --input "<path>/…AİDAT LİSTESİ.xlsx" [--sheet "2026 - 2027"] [--out <path>]
 *
 * Telefon: geçmiş kaynaklardan (leads/students csv, DATA KAYIT xlsx, .vcf, _chat.txt)
 * skorlanır; güven etiketi ile yazılır. Orta/düşük güven insan tarafından doğrulanmalı.
 */
import fs from "node:fs";
import path from "node:path";
import { fileURLToPath } from "node:url";
import xlsx from "xlsx";
import { normalizeTurkishPhone } from "../../lib/phone";
import { toCsv } from "./lib/csv";
import { buildRosterRow } from "./lib/aidat-parse";
import {
  normalizeName, buildQuery, scorePhone, type PoolEntry, type ChatBlock,
} from "./lib/phone-recovery";

const SCRIPT_DIR = path.dirname(fileURLToPath(import.meta.url));

function arg(name: string, fallback?: string): string | undefined {
  const i = process.argv.indexOf(name);
  return i !== -1 ? process.argv[i + 1] : fallback;
}

// ---- Telefon havuzu: mevcut yerel kaynaklardan (kişisel veri LOG'lanmaz) ----
function loadPool(whatsappDir: string, seedOutDir: string): { pool: PoolEntry[]; chat: ChatBlock[] } {
  const pool: PoolEntry[] = [];
  const push = (name: unknown, rawPhone: unknown, source: string) => {
    const phone = normalizeTurkishPhone(String(rawPhone ?? ""));
    const nm = normalizeName(String(name ?? ""));
    if (phone && nm) pool.push({ name: nm, phone, source });
  };

  // seed CSV'leri (varsa)
  for (const [file, cols] of [["students.csv", ["full_name", "parent_name"]], ["leads.csv", ["child_name", "parent_name"]]] as const) {
    const p = path.join(seedOutDir, file);
    if (!fs.existsSync(p)) continue;
    const lines = fs.readFileSync(p, "utf8").split(/\r?\n/).filter(Boolean);
    const header = lines[0].split(",");
    for (const line of lines.slice(1)) {
      const cells = line.split(","); // out CSV'leri basit; tırnaklı alan yoksa yeterli
      const rec: Record<string, string> = {};
      header.forEach((h, i) => (rec[h] = cells[i] ?? ""));
      for (const c of cols) push(rec[c], rec.phone, `${file}/${c}`);
    }
  }

  // DATA KAYIT xlsx'leri: col1 ad, col4 anne, col5 baba, col6 telefon
  if (fs.existsSync(whatsappDir)) {
    for (const f of fs.readdirSync(whatsappDir).filter((x) => /DATA KAYIT.*\.xlsx$/i.test(x))) {
      const wb = xlsx.readFile(path.join(whatsappDir, f));
      const rows = xlsx.utils.sheet_to_json<unknown[]>(wb.Sheets[wb.SheetNames[0]], { header: 1, defval: "" });
      for (let i = 4; i < rows.length; i++) {
        const r = rows[i]; if (!r) continue;
        push(r[1], r[6], f); push(r[4], r[6], f); push(r[5], r[6], f);
      }
    }
    // .vcf kartları: FN + TEL
    for (const f of fs.readdirSync(whatsappDir).filter((x) => x.toLowerCase().endsWith(".vcf"))) {
      const t = fs.readFileSync(path.join(whatsappDir, f), "utf8");
      const fn = (t.match(/FN[^:]*:(.+)/) || [])[1] || f.replace(/\.vcf$/i, "");
      for (const m of t.matchAll(/TEL[^:]*:(.+)/g)) push(fn, m[1], `vcf`);
    }
  }

  // _chat.txt: mesaj bloklarını çıkar, blok metni + içindeki telefonlar
  const chat: ChatBlock[] = [];
  const chatPath = path.join(whatsappDir, "_chat.txt");
  if (fs.existsSync(chatPath)) {
    const lines = fs.readFileSync(chatPath, "utf8").split(/\r?\n/);
    const lineRe = /^\[[^\]]+\]\s*([^:]+):\s*(.*)$/;
    let cur: { sender: string; text: string } | null = null;
    const blocks: { sender: string; text: string }[] = [];
    for (const line of lines) {
      const m = line.match(lineRe);
      if (m) { if (cur) blocks.push(cur); cur = { sender: m[1], text: m[2] }; }
      else if (cur) cur.text += "\n" + line;
    }
    if (cur) blocks.push(cur);
    for (const b of blocks) {
      const phones = new Set<string>();
      for (const m of b.text.matchAll(/(?:\+?90|0)?\s*5\d{2}[\s\d]{7,}/g)) {
        const p = normalizeTurkishPhone(m[0]); if (p) phones.add(p);
      }
      const sp = normalizeTurkishPhone(b.sender); if (sp) phones.add(sp);
      if (phones.size) chat.push({ text: normalizeName(`${b.sender} ${b.text}`), phones: [...phones] });
    }
  }
  return { pool, chat };
}

function main() {
  const input = arg("--input");
  if (!input) {
    console.error('Kullanım: npx tsx scripts/seed/parse-aidat.ts --input "<path>/…AİDAT LİSTESİ.xlsx" [--sheet "2026 - 2027"] [--out <path>]');
    process.exit(1);
  }
  if (!fs.existsSync(input)) { console.error(`Girdi bulunamadı: ${input}`); process.exit(1); }
  const sheet = arg("--sheet", "2026 - 2027")!;
  const out = arg("--out", path.join(SCRIPT_DIR, "out", "students-2026-2027.csv"))!;

  const wb = xlsx.readFile(input);
  if (!wb.Sheets[sheet]) { console.error(`Sayfa bulunamadı: "${sheet}". Mevcut: ${wb.SheetNames.join(", ")}`); process.exit(1); }
  const rows = xlsx.utils.sheet_to_json<unknown[]>(wb.Sheets[sheet], { header: 1, defval: "" });

  const repoRoot = path.join(SCRIPT_DIR, "..", "..");
  const { pool, chat } = loadPool(path.join(repoRoot, "..", "whatsapp_export"), path.join(SCRIPT_DIR, "out"));

  const outRows: (string | number | null)[][] = [];
  let matched = 0;
  for (let i = 1; i < rows.length; i++) {
    const rr = buildRosterRow(rows[i], i + 1);
    if (!rr) continue;
    const q = buildQuery(rr.full_name, rr.parent_name);
    const m = scorePhone(q, pool, chat);
    if (m) matched++;
    outRows.push([
      rr.source_row, rr.full_name, rr.parent_name,
      m?.phone ?? "", m?.confidence ?? "", m?.source ?? "",
      rr.program ?? "", rr.program_note, rr.season,
      rr.aidat_monthly ?? "", rr.servis_monthly ?? "", rr.kitap_amount ?? "", rr.note,
    ]);
  }

  fs.mkdirSync(path.dirname(out), { recursive: true });
  fs.writeFileSync(out, toCsv(
    ["source_row", "full_name", "parent_name", "phone", "phone_confidence", "phone_source",

// scripts/seed/lib/aidat-upload.ts
/**
 * Gözden geçirilmiş CSV satırlarını DB insert PLANINA çevirir — saf fonksiyon, DB YOK.
 * Para: TL girilir, taksitler kuruş üzerinden bölünür (lib/installments), TL olarak
 * saklanır (actions/payments.ts ile birebir aynı boru hattı).
 */
import { tlToKurus, kurusToTl } from "../../lib/money";
import { generateInstallments } from "../../lib/installments";
import { PROGRAMS } from "../../lib/constants";
import { normalizeTurkishPhone } from "../../lib/phone";
import { parseTl } from "./aidat-parse";

export const AIDAT_FIRST_DUE = "2026-09-05";
export const AIDAT_COUNT = 10;

export type InstallmentDraftRow = { due_date: string; amount: number; status: "bekliyor" };
export type PlanDraft = {
  item_type: "aidat" | "servis" | "diger";
  total_amount: number;
  installment_count: number;
  installments: InstallmentDraftRow[];
};
export type StudentInsert = { full_name: string; parent_name: string; phone: string; season: string; program: string };
export type ImportUnit = { source_row: number; student: StudentInsert; plans: PlanDraft[] };
export type SkipReport = { row: number; reason: string };

function makePlan(item_type: PlanDraft["item_type"], monthlyTl: number, count: number, firstDue: string): PlanDraft {
  const totalKurus = tlToKurus(monthlyTl * count);
  const drafts = generateInstallments(totalKurus, count, firstDue);
  return {
    item_type,
    total_amount: kurusToTl(totalKurus),
    installment_count: count,
    installments: drafts.map((d) => ({ due_date: d.dueDate, amount: kurusToTl(d.amountKurus), status: "bekliyor" })),
  };
}

export function buildImportPlan(objects: Record<string, string>[]): { units: ImportUnit[]; skipped: SkipReport[] } {
  const units: ImportUnit[] = [];
  const skipped: SkipReport[] = [];

  objects.forEach((o, idx) => {
    const rowNo = Number(o.source_row) || idx + 2;
    const full_name = (o.full_name ?? "").trim();
    if (!full_name) return skipped.push({ row: rowNo, reason: "full_name boş" });
    const phone = normalizeTurkishPhone(o.phone ?? "");
    if (!phone) return skipped.push({ row: rowNo, reason: "geçersiz/boş telefon" });
    const parent_name = (o.parent_name ?? "").trim();
    if (!parent_name) return skipped.push({ row: rowNo, reason: "parent_name boş" });
    const program = (o.program ?? "").trim();
    if (!(program in PROGRAMS)) return skipped.push({ row: rowNo, reason: `geçersiz/boş program: "${program}"` });
    const season = (o.season ?? "").trim();
    if (!season) return skipped.push({ row: rowNo, reason: "season boş" });

    const plans: PlanDraft[] = [];
    const aidat = parseTl(o.aidat_monthly);
    if (aidat) plans.push(makePlan("aidat", aidat, AIDAT_COUNT, AIDAT_FIRST_DUE));
    const servis = parseTl(o.servis_monthly);
    if (servis) plans.push(makePlan("servis", servis, AIDAT_COUNT, AIDAT_FIRST_DUE));
    const kitap = parseTl(o.kitap_amount);
    if (kitap) plans.push(makePlan("diger", kitap, 1, AIDAT_FIRST_DUE));

    units.push({ source_row: rowNo, student: { full_name, parent_name, phone, season, program }, plans });
  });

  return { units, skipped };
}
```

- [ ] **Step 1: Write the CLI**

```ts
// scripts/seed/upload-aidat.ts
/**
 * out/students-2026-2027.csv → Supabase (service role). Varsayılan DRY-RUN.
 * Yalnız insan onayından SONRA ve doğru ortama (yerel) çalıştırılır — spec §9.3.
 *
 * Kullanım:
 *   npx tsx scripts/seed/upload-aidat.ts --dry-run
 *   npx tsx scripts/seed/upload-aidat.ts            # gerçek yükleme (yerel)
 *   npx tsx scripts/seed/upload-aidat.ts --students <path>
 *
 * Yeniden çalıştırma güvenli: öğrenci telefon+ad ile, plan season+item_type ile atlanır.
 */
import fs from "node:fs";
import path from "node:path";
import crypto from "node:crypto";
import { fileURLToPath } from "node:url";
import dotenv from "dotenv";
import ws from "ws";
import { createClient, type SupabaseClient } from "@supabase/supabase-js";
import { csvToObjects } from "./lib/csv";
import { buildImportPlan, type ImportUnit } from "./lib/aidat-upload";

(globalThis as unknown as { WebSocket: typeof ws }).WebSocket = ws;

const SCRIPT_DIR = path.dirname(fileURLToPath(import.meta.url));
const envLocalPath = path.join(SCRIPT_DIR, "..", "..", ".env.local");
if (fs.existsSync(envLocalPath)) dotenv.config({ path: envLocalPath });

function flag(name: string) { return process.argv.includes(name); }
function value(name: string) { const i = process.argv.indexOf(name); return i !== -1 ? process.argv[i + 1] : null; }

function studentKey(phone: string, fullName: string) {
  return `${phone}::${fullName.trim().toLocaleLowerCase("tr")}`;
}

async function main() {
  const dryRun = flag("--dry-run");
  const studentsPath = value("--students") ?? path.join(SCRIPT_DIR, "out", "students-2026-2027.csv");
  if (!fs.existsSync(studentsPath)) { console.error(`CSV bulunamadı: ${studentsPath}`); process.exit(1); }

  const { units, skipped } = buildImportPlan(csvToObjects(fs.readFileSync(studentsPath, "utf8")));

  console.log("=== upload-aidat özet ===");
  console.log(`Mod: ${dryRun ? "DRY RUN (DB'ye yazılmayacak)" : "GERÇEK YÜKLEME (yerel)"}`);
  console.log(`Öğrenci: yüklenecek=${units.length} atlanacak=${skipped.length}`);
  for (const s of skipped) console.log(`  [satır ${s.row}] ${s.reason}`);
  const planCount = units.reduce((a, u) => a + u.plans.length, 0);
  const instCount = units.reduce((a, u) => a + u.plans.reduce((b, p) => b + p.installments.length, 0), 0);
  console.log(`Plan: ${planCount} | Taksit: ${instCount}`);

  if (dryRun) { console.log("\nDry-run: DB'ye hiçbir şey yazılmadı."); return; }

  const url = process.env.NEXT_PUBLIC_SUPABASE_URL;
  const serviceKey = process.env.SUPABASE_SECRET_KEY;
  if (!url || !serviceKey) { console.error("NEXT_PUBLIC_SUPABASE_URL ve SUPABASE_SECRET_KEY gerekli."); process.exit(1); }
  const db: SupabaseClient = createClient(url, serviceKey, { auth: { persistSession: false } });

  // Önceki seed uyarısı (engellemez)
  const { data: prev } = await db.from("events").select("id").eq("event_type", "seed_import").eq("entity_type", "student").limit(1);
  if (prev && prev.length) console.warn("UYARI: daha önce bir student seed_import event'i var — telefon+ad ve season+item_type kontrolleri tekrarı önler.");

  // Mevcut öğrenci anahtarları
  const { data: existingStudents, error: sErr } = await db.from("students").select("id, phone, full_name");
  if (sErr) throw new Error(`öğrenciler okunamadı: ${sErr.message}`);
  const keyToId = new Map<string, string>();
  for (const r of (existingStudents ?? []) as { id: string; phone: string; full_name: string }[]) {
    keyToId.set(studentKey(r.phone, r.full_name), r.id);
  }

  let insertedStudents = 0, insertedPlans = 0, insertedInst = 0;
  for (const u of units) {
    let studentId = keyToId.get(studentKey(u.student.phone, u.student.full_name));
    if (!studentId) {
      const { data, error } = await db.from("students").insert(u.student).select("id").single();
      if (error) throw new Error(`student insert (satır ${u.source_row}): ${error.message}`);
      studentId = data.id as string;
      insertedStudents++;
    }
    // Bu öğrencinin bu season'daki mevcut planlarının item_type'ları
    const { data: existingPlans } = await db.from("payment_plans")
      .select("item_type").eq("student_id", studentId).eq("season", u.student.season);
    const have = new Set((existingPlans ?? []).map((p: { item_type: string }) => p.item_type));

    for (const plan of u.plans) {
      if (have.has(plan.item_type)) continue; // yeniden çalıştırma: atla
      const { data: planRow, error: pErr } = await db.from("payment_plans").insert({
        student_id: studentId, season: u.student.season, item_type: plan.item_type,
        total_amount: plan.total_amount, installment_count: plan.installment_count,
      }).select("id").single();
      if (pErr) throw new Error(`plan insert (satır ${u.source_row}, ${plan.item_type}): ${pErr.message}`);
      insertedPlans++;
      const instRows = plan.installments.map((i) => ({ plan_id: planRow.id, due_date: i.due_date, amount: i.amount, status: i.status }));
      const { error: iErr } = await db.from("installments").insert(instRows);
      if (iErr) throw new Error(`installments insert (satır ${u.source_row}, ${plan.item_type}): ${iErr.message}`);
      insertedInst += instRows.length;
    }
  }

  // Tek işaret event'i (tarihi veri — satır başına event yok)
  await db.from("events").insert({
    entity_type: "student", entity_id: crypto.randomUUID(), event_type: "seed_import",
    payload: { seed: true, source: "aidat-2026-2027", count: insertedStudents },
  });

  console.log(`\nYüklendi — öğrenci: ${insertedStudents}, plan: ${insertedPlans}, taksit: ${insertedInst}`);
}

main().catch((e) => { console.error(e); process.exit(1); });
```
